cmake_minimum_required(VERSION 3.20)
project(DeadSpaceMobileDesktopResearch VERSION 0.1.0 LANGUAGES CXX)

# Infrastructure only: there is no recovered gameplay engine at this stage.
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(DS_BUILD_INPUT_PROBE "Build SDL2 keyboard/mouse diagnostic program" ON)
option(DS_FETCH_SDL2 "Download and build pinned SDL2 if not installed (on CI, not the assistant VM)" OFF)
option(DS_ENABLE_TESTS "Build and register deterministic input tests" ON)
option(DS_BUILD_ASSET_INSTALLER "Build user-owned APK assets/ installer with zlib" OFF)
option(DS_BUILD_M3G_READER "Build format-verified JSR-184 M3G section parser" OFF)
option(DS_BUILD_IM_MESH_READER "Build native IronMonkey geometry-only mesh exporter" OFF)

# These functions are independent x86-64 translations of 29 validated
# original ARM game JNI constant returns; no guessed gameplay dispatch.
add_library(ds_original_native_abi STATIC src/original_native_abi.cpp src/original_jni_exports.cpp src/original_frame_dispatch.cpp src/original_game_entry.cpp src/original_timing_math.cpp src/original_clock_source.cpp src/original_stage_zero_setup.cpp src/original_stage_zero_followup.cpp src/original_stage_zero_type_two.cpp src/original_record_postalloc.cpp src/original_runtime_globals.cpp src/original_allocator_call.cpp src/original_stage_zero_allocating.cpp src/original_registry_lookup.cpp src/original_secondary_allocator.cpp src/original_secondary_postprocess.cpp src/original_secondary_pipeline.cpp src/original_object_constructor.cpp)
target_include_directories(ds_original_native_abi PUBLIC include)

# Native executable must report missing original game code rather than run
# a diagnostic pretending to be playable. It links the same recovered ABI.
add_executable(deadspace_native apps/deadspace_native.cpp)
target_link_libraries(deadspace_native PRIVATE ds_original_native_abi)

add_library(ds_port_input STATIC src/input.cpp)
add_library(ds_port_timing STATIC src/timing.cpp)
target_include_directories(ds_port_timing PUBLIC include)

add_library(ds_port_apk_probe STATIC src/apk_probe.cpp)
target_include_directories(ds_port_apk_probe PUBLIC include)
add_executable(ds_apk_probe apps/apk_probe.cpp)
target_link_libraries(ds_apk_probe PRIVATE ds_port_apk_probe)
target_include_directories(ds_port_input PUBLIC include)
if(MSVC)
    target_compile_options(ds_port_input PRIVATE /W4)
    target_compile_options(ds_port_timing PRIVATE /W4)
    target_compile_options(ds_port_apk_probe PRIVATE /W4)
else()
    target_compile_options(ds_port_input PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(ds_port_timing PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(ds_port_apk_probe PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(DS_BUILD_INPUT_PROBE)
    find_package(SDL2 CONFIG QUIET)
    if(NOT TARGET SDL2::SDL2 AND DS_FETCH_SDL2)
        include(FetchContent)
        set(SDL_TEST OFF CACHE BOOL "SDL tests" FORCE)
        set(SDL_SHARED OFF CACHE BOOL "SDL shared library" FORCE)
        set(SDL_STATIC ON CACHE BOOL "SDL static library" FORCE)
        FetchContent_Declare(ds_sdl2
            URL "https://www.libsdl.org/release/SDL2-2.30.12.tar.gz"
            URL_HASH "SHA256=ac356ea55e8b9dd0b2d1fa27da40ef7e238267ccf9324704850d5d47375b48ea"
            DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
        FetchContent_MakeAvailable(ds_sdl2)
    endif()
    if(TARGET SDL2::SDL2)
        add_executable(ds_input_probe apps/input_probe.cpp)
        target_link_libraries(ds_input_probe PRIVATE ds_port_input ds_port_timing SDL2::SDL2)
        if(TARGET SDL2::SDL2main)
            target_link_libraries(ds_input_probe PRIVATE SDL2::SDL2main)
        endif()
        message(STATUS "SDL2 found; building ds_input_probe (diagnostic, NOT a game)")
    else()
        message(STATUS "SDL2 unavailable; skipping optional input probe. Core and tests still build.")
    endif()
endif()


# The installer is opt-in so the existing lightweight diagnostic build does
# not gain extra dependencies. ZIP inflate/CRC always runs on the user's PC.
if(DS_BUILD_ASSET_INSTALLER)
    find_package(ZLIB REQUIRED)
    add_library(ds_port_assets STATIC src/asset_install.cpp)
    target_include_directories(ds_port_assets PUBLIC include)
    target_link_libraries(ds_port_assets PUBLIC ds_port_apk_probe ZLIB::ZLIB)
    if(MSVC)
        target_compile_options(ds_port_assets PRIVATE /W4)
    else()
        target_compile_options(ds_port_assets PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    add_executable(ds_asset_install apps/asset_install.cpp)
    target_link_libraries(ds_asset_install PRIVATE ds_port_assets)
endif()
