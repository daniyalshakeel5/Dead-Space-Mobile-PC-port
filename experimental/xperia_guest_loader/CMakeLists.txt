cmake_minimum_required(VERSION 3.20)
project(xperia_guest_elf32_loader LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
add_library(xperia_guest_elf32 STATIC guest_elf32.cpp)
target_include_directories(xperia_guest_elf32 PUBLIC .)
add_executable(xperia_guest_elf32_tests guest_elf32_tests.cpp)
target_link_libraries(xperia_guest_elf32_tests PRIVATE xperia_guest_elf32)
enable_testing()
add_test(NAME xperia_elf32_relative_and_external_relocations
         COMMAND xperia_guest_elf32_tests)

# Original-free, ARM AAPCS libc import service dispatcher.
add_library(xperia_guest_heap STATIC guest_heap.cpp)
target_link_libraries(xperia_guest_heap PUBLIC xperia_guest_elf32)
target_include_directories(xperia_guest_heap PUBLIC .)
add_library(xperia_guest_filesystem STATIC guest_filesystem.cpp)
target_include_directories(xperia_guest_filesystem PUBLIC .)
add_library(xperia_guest_libc_bridge STATIC guest_libc_bridge.cpp guest_jni_vm.cpp guest_bionic_data.cpp
    guest_threads.cpp guest_c_runtime.cpp)
target_link_libraries(xperia_guest_libc_bridge PUBLIC xperia_guest_elf32 xperia_guest_heap xperia_guest_filesystem)
target_include_directories(xperia_guest_libc_bridge PUBLIC .)
add_executable(xperia_guest_libc_bridge_tests guest_libc_bridge_tests.cpp)
target_link_libraries(xperia_guest_libc_bridge_tests PRIVATE xperia_guest_libc_bridge)
add_test(NAME xperia_guest_bionic_libc_memory_strings COMMAND xperia_guest_libc_bridge_tests)

add_executable(xperia_guest_heap_tests guest_heap_tests.cpp)
target_link_libraries(xperia_guest_heap_tests PRIVATE xperia_guest_libc_bridge)
add_test(NAME xperia_guest_bionic_heap_allocation COMMAND xperia_guest_heap_tests)

add_executable(xperia_guest_filesystem_tests guest_filesystem_tests.cpp)
target_link_libraries(xperia_guest_filesystem_tests PRIVATE xperia_guest_filesystem)
add_test(NAME xperia_guest_original_asset_io_sandbox COMMAND xperia_guest_filesystem_tests)

# Fast Windows tests: no Dynarmic/Boost/SDL dependency.
add_executable(xperia_guest_jni_vm_tests guest_jni_vm_tests.cpp)
target_link_libraries(xperia_guest_jni_vm_tests PRIVATE xperia_guest_libc_bridge)
add_test(NAME xperia_guest_jni_classes_audio_and_device_profile
    COMMAND xperia_guest_jni_vm_tests)
add_executable(xperia_guest_bionic_data_tests guest_bionic_data_tests.cpp)
target_link_libraries(xperia_guest_bionic_data_tests PRIVATE xperia_guest_libc_bridge)
add_test(NAME xperia_guest_bionic_elf_global_objects
    COMMAND xperia_guest_bionic_data_tests)

# Independent compressed-texture correctness tests run on Windows without JIT.
add_executable(xperia_guest_texture_decoder_tests
    ../xperia_armjit/guest_texture_decoder_tests.cpp
    ../xperia_armjit/guest_texture_decoder.cpp
    ../xperia_armjit/third_party/powervr/PVRTDecompress.cpp)
target_include_directories(xperia_guest_texture_decoder_tests
    PRIVATE ../xperia_armjit)
add_test(NAME xperia_guest_ATC_PVRTC_decoding
    COMMAND xperia_guest_texture_decoder_tests)

add_executable(xperia_guest_threads_tests guest_threads_tests.cpp)
target_link_libraries(xperia_guest_threads_tests PRIVATE xperia_guest_libc_bridge)
add_test(NAME xperia_guest_pthread_abi_lifecycle COMMAND xperia_guest_threads_tests)

add_executable(xperia_guest_pthread_services_tests guest_pthread_services_tests.cpp)
target_link_libraries(xperia_guest_pthread_services_tests PRIVATE xperia_guest_libc_bridge)
add_test(NAME xperia_guest_pthread_cond_tls_and_mutex_attributes
    COMMAND xperia_guest_pthread_services_tests)

add_executable(xperia_guest_c_runtime_tests guest_c_runtime_tests.cpp)
target_link_libraries(xperia_guest_c_runtime_tests PRIVATE xperia_guest_libc_bridge)
add_test(NAME xperia_arm32_guest_safe_strings_printf_and_logs
    COMMAND xperia_guest_c_runtime_tests)
